KALENDER-REGELN & AUTOMATISCHE SPERREN --- var d1 = document.getElementById('fl-date1'); var d2 = document.getElementById('fl-date2'); if (d1 && d2) { // Standarddaten setzen d1.value = "2026-11-15"; d2.value = "2026-11-25"; d2.min = "2026-11-15"; // Sperrt sofort alle vergangenen Tage vor dem Hinflug // Live-Sperre: Hinflug-Änderung sperrt Rückflug-Tage davor d1.addEventListener('change', function() { d2.min = this.value; if (d2.value && d2.value < this.value) { d2.value = this.value; // Zieht den Rückflug automatisch hoch } }); } // --- 2. NUR HINFLUG (ONE WAY) AUTOMATIK --- var oneWayCheck = document.getElementById('fl-oneway'); if (oneWayCheck && d2 && d1) { oneWayCheck.addEventListener('change', function() { if (this.checked) { d2.disabled = true; d2.value = ""; } else { d2.disabled = false; d2.value = "2026-11-25"; d2.min = d1.value; } }); } // --- 3. DIE FLUGSUCHMASCHINE (START BEI KLICK) --- var searchBtn = document.getElementById('btn-search-flights'); if (searchBtn && d1 && d2 && oneWayCheck) { searchBtn.addEventListener('click', function() { var fromInput = document.getElementById('fl-from').value; var toInput = document.getElementById('fl-to').value; var includeLowCost = document.getElementById('fl-lowcost').checked; // Hilfsfunktion: Filtert den Code aus den Klammern (z.B. "Frankfurt (FRA)" -> "FRA") var extractAirportCode = function(text) { var match = text.match(/\(([^)]+)\)/); return match ? match[1] : text; }; var fromCode = extractAirportCode(fromInput).toUpperCase().trim(); var toCode = extractAirportCode(toInput).toUpperCase().trim(); if (!fromCode || !toCode || fromCode.length !== 3 || toCode.length !== 3) { alert("Bitte wähle Flughäfen aus der Vorschlagsliste aus (z.B. Frankfurt (FRA))!"); return; } // KORRIGIERTER GOOGLE FLIGHTS LINK MIT ECHTEM PFAD var googleQuery = "flights from " + fromCode + " to " + toCode + " on " + d1.value; if (!oneWayCheck.checked && d2.value) { googleQuery += " return " + d2.value; } else { googleQuery += " one way"; } var googleUrl = "https://google.com" + encodeURIComponent(googleQuery) + "&hl=de"; window.open(googleUrl, '_blank'); // KORRIGIERTER SKYSCANNER LINK MIT ECHTEM PFAD var dDate = d1.value.replace(/-/g, '').substring(2); // Macht "261115" var skyscannerUrl = "https://skyscanner.de" + fromCode.toLowerCase() + "/" + toCode.toLowerCase() + "/" + dDate; if (!oneWayCheck.checked && d2.value) { var rDate = d2.value.replace(/-/g, '').substring(2); // Macht "261125" skyscannerUrl += "/" + rDate + "/?adults=1&rtn=1&language=de"; } else { skyscannerUrl += "/?adults=1&rtn=0&language=de"; } window.open(skyscannerUrl, '_blank'); // BILLIGFLIEGER ERWEITERUNG if (includeLowCost) { window.open("https://cebupacificair.com", "_blank"); window.open("https://trip.com", "_blank"); } }); }
