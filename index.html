============================================================================= # AB HIER STARTET DIE AUTOMATISCHE LOGIK (KEINE ÄNDERUNGEN NÖTIG) # ============================================================================= import smtplib from email.mime.multipart import MIMEMultipart from email.mime.text import MIMEText from google.colab import userdata # Sicherer Abruf deiner Zugangsdaten aus dem Colab-Schlüsselspeicher try: ABSENDER_EMAIL = userdata.get('ABSENDER_EMAIL') PASSWORT = userdata.get('PASSWORT') EMPFAENGER_EMAIL = ABSENDER_EMAIL except Exception: print("❌ FEHLER: Bitte speichere links unter dem Schlüssel-Symbol (Secrets) 'ABSENDER_EMAIL' und 'PASSWORT' ab!") print("Verarbeite erweiterte Suchkriterien für optimale Reichweite...") # Berechnung der Flugkosten basierend auf der Transfer-Auswahl if SEPARATER_TRANSFER and ZWISCHENSTOPS != "direkt": flug_gesamt = (PREIS_STRECKE_1 + PREIS_STRECKE_2) * PERSONEN flug_info = "Kombiniert (St. 1: %.2f EUR, St. 2: %.2f EUR)" % (PREIS_STRECKE_1, PREIS_STRECKE_2) else: flug_gesamt = PREIS_STRECKE_1 * PERSONEN flug_info = "Standard / Anschlussflug" # Gesamtkalkulation gesamtkosten = flug_gesamt + HOTEL_PREIS_GESAMT gesamt_budget = MAX_BUDGET_FLUG + MAX_BUDGET_HOTEL # Text-Formatierungen flug_text = "%.2f EUR" % flug_gesamt hotel_text = "%.2f EUR" % HOTEL_PREIS_GESAMT gesamtkosten_text = "%.2f EUR" % gesamtkosten budget_flug_text = "%.2f EUR" % MAX_BUDGET_FLUG budget_hotel_text = "%.2f EUR" % MAX_BUDGET_HOTEL gesamt_budget_text = "%.2f EUR" % gesamt_budget # Detail-Prüfungen für das E-Mail-Layout status_flug = "✅ OK" if flug_gesamt <= MAX_BUDGET_FLUG else "❌ Drüber" status_hotel = "✅ OK" if HOTEL_PREIS_GESAMT <= MAX_BUDGET_HOTEL else "❌ Drüber" if gesamtkosten <= gesamt_budget: status_farbe = "#2ecc71" status_titel = "Budget eingehalten!" rest = gesamt_budget - gesamtkosten status_text = "Hervorragend! Deine Reisekombination liegt im Limit. Gesamt-Restbudget: %.2f EUR." % rest else: status_farbe = "#e74c3c" status_titel = "Budget überschritten!" ueber = gesamtkosten - gesamt_budget status_text = "Achtung! Deine ausgewählten Tarife überschreiten das Gesamtbudget um %.2f EUR." % ueber # Korrigierte Link-Generierung für Kiwi (Jahr-Monat-Tag zu Tag/Monat/Jahr) parts = HIN_DATUM.split("-") datum_kiwi = parts[2] + "/" + parts[1] + "/" + parts[0] such_link = "https://kiwi.com" + VON.lower() + "/" + NACH.lower() + "/" + datum_kiwi # ========================================== # GENERIERUNG DES LAYOUTS FÜR WEB.DE # ========================================== html_zeilen = [ "
